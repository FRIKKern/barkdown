test('actual Windows native handle observations preserve bytes and refuse unsafe file states',{skip:process.platform!=='win32'},async()=>{
 const require=createRequire(import.meta.url),{powershell,systemRoot,systemTool}=require('../app/main/windows-system-tools');
 const root=fs.mkdtempSync(path.join(os.tmpdir(),'Barkdown C103 handle Chø ')),outside=fs.mkdtempSync(path.join(os.tmpdir(),'Barkdown C103 outside '));fs.mkdirSync(path.join(root,'originals'));

 const key='originals/Chø 日本語.png',file=path.join(root,key),bytes=crypto.randomBytes(4096);fs.writeFileSync(file,bytes);const input={...request,root:fs.realpathSync(root),rootIdentity:identity,key,maxBytes:4096},hash=crypto.createHash('sha256').update(bytes).digest('hex');const observations=[];
 const observe=(label,change={})=>{const r=observeWindowsMediaFile({...input,...change});observations.push({label,result:r});fs.writeFileSync(path.join(root,'observations.json'),JSON.stringify({status:'running',root,outside,observations},null,2)+'\n');for(const flag of ['collectionComplete','reconciled','deletionAllowed','recoveryAllowed'])assert.equal(r[flag],false);assert.deepEqual(fs.readFileSync(file),bytes);return r;};
 let r=observe('Unicode original');assert.equal(r.file.state,'readable');assert.equal(r.file.sha256,hash);assert.equal(r.file.bytes,4096);assert.equal(r.file.identity.inode,fs.statSync(file,{bigint:true}).ino.toString());assert.equal(r.file.identity.device,identity.volume);
 r=observe('case alias',{key:'originals/chØ 日本語.png'});assert.equal(r.file.state,'unsupported');assert.equal(r.diagnostics.reason,'file_path_alias');assert.equal(r.file.sha256,null);
 const empty=path.join(root,'empty.bin');fs.writeFileSync(empty,Buffer.alloc(0));r=observe('empty original',{key:'empty.bin',maxBytes:1});assert.equal(r.file.state,'readable');assert.equal(r.file.bytes,0);assert.equal(r.file.sha256,crypto.createHash('sha256').update(Buffer.alloc(0)).digest('hex'));

 r=observe('byte limit',{maxBytes:4095});assert.equal(r.file.state,'unsupported');assert.equal(r.diagnostics.reason,'byte_bound_exhausted');assert.equal(r.file.sha256,null);
 r=observe('missing file',{key:'originals/missing.png'});assert.equal(r.file.state,'absent');assert.equal(r.diagnostics.win32Error,2);
 r=observe('missing nested directory',{key:'missing/child.png'});assert.equal(r.file.state,'absent');
 r=observe('missing root',{root:path.join(root,'missing-root')});assert.equal(r.file.state,'unreadable');

 const sensitive=path.join(outside,'sensitive.bin');fs.writeFileSync(sensitive,'owned outside control');fs.symlinkSync(outside,path.join(root,'junction'),'junction');r=observe('outside junction',{key:'junction/sensitive.bin'});assert.equal(r.file.state,'unsupported');assert.equal(r.diagnostics.reason,'unsupported_directory');assert.equal(r.file.sha256,null);assert.equal(fs.readFileSync(sensitive,'utf8'),'owned outside control');
