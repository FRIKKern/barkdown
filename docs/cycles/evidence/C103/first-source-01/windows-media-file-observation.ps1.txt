using System;
using System.IO;
using System.Text;

using System.Collections.Generic;
using System.ComponentModel;
using System.Runtime.InteropServices;

  [StructLayout(LayoutKind.Sequential)] struct Stamp { public uint low, high; }
  [StructLayout(LayoutKind.Sequential)] struct Info {
    public uint attributes; public Stamp creation, access, write;

  }
  [DllImport("kernel32.dll", CharSet=CharSet.Unicode, SetLastError=true)] static extern SafeFileHandle CreateFileW(string name,uint access,uint share,IntPtr security,uint disposition,uint flags,IntPtr template);
  [DllImport("kernel32.dll",SetLastError=true)] static extern bool GetFileInformationByHandle(SafeFileHandle h,out Info info);
  [DllImport("kernel32.dll",CharSet=CharSet.Unicode,SetLastError=true)] static extern uint GetFinalPathNameByHandleW(SafeFileHandle h,StringBuilder path,uint size,uint flags);

  static ulong Index(Info i) { return ((ulong)i.indexHigh<<32)|i.indexLow; }
  static ulong Size(Info i) { return ((ulong)i.sizeHigh<<32)|i.sizeLow; }

    var result=new Dictionary<string,object>{{"state","unsupported"},{"reason","unobserved"},{"win32Error",null},{"bytes",null},{"sha256",null},{"identity",null},{"finalPath",null}};

    try {

      // This is a scoped offline observation, not a runtime or whole-tree lease.
      for(int n=0;n<parts.Length;n++) {

        if(!String.Equals(Final(h),current,StringComparison.Ordinal)){result["reason"]="directory_path_alias";return result;}
        if(n==0&&(info.volume!=expectedVolume||Index(info)!=expectedInode))throw new ArgumentException("Root identity differs");
        if(n==0)rootBound=true;
        before.Add(info);names.Add(current);if(n<parts.Length-1)current+="\\"+parts[n];
      }
      string file=current+"\\"+parts[parts.Length-1];var fh=CreateFileW(file,0x80000000u,1,IntPtr.Zero,3,0x00200000u,IntPtr.Zero);handles.Add(fh);

      }

      result["identity"]=new Dictionary<string,object>{{"device",first.volume.ToString()},{"inode",Index(first).ToString()},{"links",(int)first.links}};return result;
    } catch(Win32Exception e) {result["win32Error"]=e.NativeErrorCode;result["state"]=(missingOpen&&(e.NativeErrorCode==2||e.NativeErrorCode==3))?"absent":"unreadable";result["reason"]="native_open_or_metadata_refused";return result;}
    finally {for(int n=handles.Count-1;n>=0;n--)handles[n].Dispose();}
  }
}
